    55→// ---------- hbar with hover-expansion fisheye lens (US-023) ----------
    56→
    57→const LENS_MAG = 1.7; // peak magnification at the lens centre
    58→const LENS_SIGMA = 46; // falloff radius in px (Gaussian)
    59→
    60→/**
    61→ * 1-D fisheye over row heights: every row's magnification is
    62→ *   m(y) = 1 + (MAG-1)·exp(-(Δy/σ)²)
    63→ * heights are then rescaled to keep the chart's total height CONSTANT, so
    64→ * the base geometry outside the lens is untouched — only the local
    65→ * neighbourhood expands. The value axis (bar widths) never changes.
    66→ */
    67→function fisheyeMagnifications(centres: number[], focusY: number | null): number[] {
    68→  if (focusY == null) return centres.map(() => 1);
    69→  return centres.map((cy) => {
    70→    const d = (cy - focusY) / LENS_SIGMA;
    71→    return 1 + (LENS_MAG - 1) * Math.exp(-d * d);
    72→  });
    73→}

    75→function FisheyeHBar({ spec }: { spec: ChatChartSpec }) {
    76→  const rows = (spec.data ?? []).map((d) => ({ ...d, color: hex(d.color) }));
    77→  const W = 460;
    78→  const H = Math.max(150, rows.length * 34 + 10);
    79→  const labelW = 150;
    80→  const valueW = 44;
    81→  const barX = labelW + 8;
    82→  const barMax = W - barX - valueW - 8;
    83→
    84→  const [focusY, setFocusY] = useState<number | null>(null);
    85→  const svgRef = useRef<SVGSVGElement>(null);
    86→
    87→  const maxValue = Math.max(...rows.map((r) => r.value), 1);
    88→
    89→  // Base (unfisheye'd) row geometry.
    90→  const rowH = H / rows.length;
    91→  const baseCentres = rows.map((_, i) => i * rowH + rowH / 2);
    92→
    93→  const mags = fisheyeMagnifications(baseCentres, focusY);
    94→
    95→  // Rescale heights so Σheights stays H (base chart keeps its footprint).
    96→  const sumM = mags.reduce((a, b) => a + b, 0);
    97→  const heights = mags.map((m) => (m / sumM) * H);
    98→
    99→  // Recomputed centres (cumulative) — far rows stay at their base positions.
   100→  const centres: number[] = [];
   101→  let acc = 0;
   102→  for (let i = 0; i < rows.length; i++) {
   103→    centres.push(acc + heights[i] / 2);
   104→    acc += heights[i];
   105→  }
   106→
   107→  const pointerMove = (e: React.PointerEvent<SVGSVGElement>) => {
   108→    const rect = svgRef.current?.getBoundingClientRect();
   109→    if (!rect) return;
   110→    const y = ((e.clientY - rect.top) / rect.height) * H;
   111→    setFocusY(y);
   112→  };
   113→  const pointerLeave = () => setFocusY(null);
   114→
   115→  // US-140: keyboard parity — arrows move the lens, Escape releases it.
   116→  const keyDown = (e: React.KeyboardEvent) => {
   117→    const idx = focusY == null ? -1 : Math.min(rows.length - 1, Math.max(0, Math.floor(focusY / rowH)));
   118→    if (e.key === "ArrowDown" || e.key === "ArrowUp") {
   119→      e.preventDefault();
   120→      const next = e.key === "ArrowDown" ? Math.min(rows.length - 1, idx + 1) : Math.max(0, idx < 0 ? 0 : idx - 1);
   121→      setFocusY(baseCentres[next]);
   122→    } else if (e.key === "Home") {
   123→      e.preventDefault();
   124→      setFocusY(baseCentres[0]);
   125→    } else if (e.key === "End") {
   126→      e.preventDefault();
   127→      setFocusY(baseCentres[rows.length - 1]);
   128→    } else if (e.key === "Escape") {
   129→      setFocusY(null);
   130→    }
   131→  };
   132→
   133→  const lensActive = focusY != null;
   134→
   135→  return (
   136→    <div className="mt-2">
   137→      {spec.title && <p className="text-sm font-semibold text-stone-900 mb-0.5">{spec.title}</p>}
   138→      {spec.tell && <p className="text-xs text-stone-500 mb-1.5">{spec.tell}</p>}
   139→      <svg
   140→        ref={svgRef}
   141→        viewBox={`0 0 ${W} ${H}`}
   142→        width="100%"
   143→        height={H}
   144→        role="img"
   145→        aria-label={`Bar chart, ${rows.length} rows: ${rows.map((r) => `${r.name} ${r.value}`).join(", ")}`}
   146→        tabIndex={0}
   147→        onPointerMove={pointerMove}
   148→        onPointerLeave={pointerLeave}
   149→        onKeyDown={keyDown}
   150→        className="rounded-lg outline-none focus-visible:ring-2 focus-visible:ring-emerald-600/40"
   151→      >
   152→        {/* rows — order and value axis never change; only local magnification.
   153→            Geometry transitions (90ms) smooth pointer/keyboard lens moves. */}
   154→        {rows.map((r, i) => {
   155→          const h = Math.max(6, heights[i] - 3);
   156→          const cy = centres[i];
   157→          const w = Math.max(3, (r.value / maxValue) * barMax);
   158→          // US-142: label + value always visible; the lens only changes size.
   159→          const labelSize = 11 + 2.5 * (mags[i] - 1);
   160→          return (
   161→            <g key={i}>
   162→              <text
   163→                x={labelW}
   164→                y={cy + labelSize * 0.35}
   165→                textAnchor="end"
   166→                fontSize={labelSize}
   167→                fill={mags[i] > 1.25 ? "#1c1917" : "#57534e"}
   168→                fontWeight={mags[i] > 1.25 ? 600 : 400}
   169→                style={{ transition: "fill 120ms" }}
   170→              >
   171→                {r.name.length > 19 ? r.name.slice(0, 18) + "…" : r.name}
   172→              </text>
   173→              <rect x={barX} y={cy - h / 2} width={barMax} height={h} rx={4} fill="#f5f5f4" opacity={lensActive ? 0.5 : 1} style={{ transition: "y 90ms ease-out, height 90ms ease-out" }} />
   174→              <rect x={barX} y={cy - h / 2} width={w} height={h} rx={4} fill={r.color} style={{ transition: "y 90ms ease-out, height 90ms ease-out, fill 120ms" }} />
   175→              <text
   176→                x={barX + w + 6}
   177→                y={cy + 4}
   178→                fontSize={labelSize}
   179→                fill={mags[i] > 1.25 ? "#1c1917" : "#78716c"}
   180→                fontWeight={mags[i] > 1.25 ? 700 : 500}
   181→                style={{ transition: "y 90ms ease-out, fill 120ms" }}
   182→              >
   183→                {r.value}
   184→              </text>
   185→            </g>
   186→          );
   187→        })}
   188→      </svg>
   189→      {rows.length >= 4 && (
   190→        <p className="text-[10px] text-stone-400 flex items-center gap-1 -mt-0.5">
   191→          <MoveVertical className="h-3 w-3" /> hover a row — or use ↑ ↓ keys — to magnify that neighbourhood
   192→        </p>
   193→      )}
   194→    </div>
   195→  );
   196→}

   295→// ---------- UML-style diagrams (uml = structure, flow = who calls whom) ----------
   296→//
   297→// Boxes + arrows built with plain SVG, same visual language as the map:
   298→// emerald = proven, amber = good guess (dashed), grey = not sure yet.
   299→// The boxes never come from the model — draw-anything.ts fills them from the
   300→// loaded store, so every label is traceable.
   301→// Layout mirrors the engine's own band hint (domain → modules → use-cases →
   302→// rules); flows layer callers above callees (longest call path).
   303→
   304→type Cert = "Fact" | "Inference" | "Hypothesis";
   305→
   306→const CERT_STYLE: Record<Cert, { fill: string; stroke: string; text: string; dash?: string }> = {
   307→  Fact: { fill: "#ecfdf5", stroke: "#059669", text: "#065f46" },
   308→  Inference: { fill: "#fffbeb", stroke: "#d97706", text: "#78350f", dash: "5 3" },
   309→  Hypothesis: { fill: "#f5f5f4", stroke: "#78716c", text: "#44403c", dash: "2 3" },
   310→};

   312→const KIND_PLAIN: Record<string, string> = {
   313→  domain: "The big picture",
   314→  module: "A code file",
   315→  usecase: "Something it does",
   316→  rule: "A rule it follows",
   317→  invariant: "Must always be true",
   318→  function: "A named thing",
   319→  step: "A step",

   322→/** Greedy word wrap — returns at most maxLines lines, last one ellipsised. */
   323→function wrap(s: string, maxChars: number, maxLines: number): string[] {
   324→  const words = s.split(/\s+/).filter(Boolean);
   325→  const out: string[] = [];
   326→  let cur = "";
   327→  for (const w of words) {
   328→    if (cur && (cur + " " + w).length > maxChars) {
   329→      out.push(cur);
   330→      cur = w;
   331→      if (out.length === maxLines) break;
   332→    } else {
   333→      cur = cur ? cur + " " + w : w;
   334→    }
   335→  }
   336→  if (out.length < maxLines && cur) out.push(cur);
   337→  if (out.length === maxLines) {
   338→    // did we drop anything?
   339→    const consumed = out.join(" ").length;
   340→    if (consumed < s.length - 1) out[maxLines - 1] = out[maxLines - 1].slice(0, Math.max(0, maxChars - 1)) + "…";
   341→  }
   342→  return out.length ? out : [s.slice(0, maxChars - 1) + "…"];
   343→}

   345→interface Box {
   346→  node: UmlNode;
   347→  row: number;
   348→  x: number;
   349→  y: number;
   350→  w: number;
   351→  h: number;
   352→  titleLines: string[];
   353→}
   354→
   355→const DG_W = 960;
   356→const DG_MARGIN = 26;
   357→const DG_GAP = 16;
   358→const ROW_GAP = 34;
   359→
   360→function DiagramSVG({ spec }: { spec: ChatChartSpec }) {
   361→  const uid = useId().replace(/[:]/g, "");
   362→  const [hover, setHover] = useState<string | null>(null);
   363→  const nodes = spec.nodes ?? [];
   364→  const links = spec.links ?? [];
   365→
   366→  // ---- rows ----
   367→  const rows: UmlNode[][] = [];
   368→  if (spec.kind === "flow") {
   369→    // depth = longest call chain from a root; callers sit above callees
   370→    const depth = new Map<string, number>();
   371→    const calc = (id: string, seen: Set<string>): number => {
   372→      if (depth.has(id)) return depth.get(id)!;
   373→      if (seen.has(id)) return 0; // cycle guard
   374→      seen.add(id);
   375→      const callers = links.filter((l) => l.to === id).map((l) => l.from);
   376→      const d = callers.length ? Math.max(...callers.map((c) => calc(c, seen) + 1)) : 0;
   377→      depth.set(id, d);
   378→      return d;
   379→    };
   380→    for (const n of nodes) calc(n.id, new Set());
   381→    const byDepth = new Map<number, UmlNode[]>();
   382→    for (const n of nodes) {
   383→      const d = depth.get(n.id) ?? 0;
   384→      byDepth.set(d, [...(byDepth.get(d) ?? []), n]);
   385→    }
   386→    [...byDepth.keys()].sort((a, b) => a - b).forEach((d) => rows.push(byDepth.get(d)!));
   387→  } else {
   388→    const bands: { kinds: string[] }[] = [
   389→      { kinds: ["domain"] },
   390→      { kinds: ["module"] },
   391→      { kinds: ["usecase"] },
   392→      { kinds: ["rule", "invariant"] },
   393→    ];
   394→    const known = new Set(bands.flatMap((b) => b.kinds));
   395→    for (const b of bands) {
   396→      const group = nodes.filter((n) => b.kinds.includes(n.kind));
   397→      if (group.length) rows.push(group);
   398→    }
   399→    const rest = nodes.filter((n) => !known.has(n.kind));
   400→    if (rest.length) rows.push(rest);
   401→  }
   402→
   403→  // ---- geometry ----
   404→  const boxes: Box[] = [];
   405→  let y = DG_MARGIN + 6;
   406→  rows.forEach((row, ri) => {
   407→    const metas = row.map((n) => {
   408→      const isLong = n.kind === "rule" || n.kind === "invariant";
   409→      const titleLines = isLong ? wrap(n.title, 30, 2) : n.title.length > 30 ? wrap(n.title, 28, 2) : [n.title];
   410→      const body = n.lines ?? [];
   411→      const bodyLines = body.map((l) => (l.length > 26 ? l.slice(0, 25) + "…" : l));
   412→      const textW = Math.max(...titleLines.map((t) => t.length), ...bodyLines.map((t) => t.length), 8);
   413→      const w = Math.min(240, Math.max(112, textW * 6.3 + 26));
   414→      const h = 8 + titleLines.length * 13.5 + (bodyLines.length ? 6 + bodyLines.length * 12.5 + (n.more ? 11 : 0) : 0) + 8;
   415→      return { node: n, titleLines, bodyLines, w, h };
   416→    });
   417→    // wrap wide rows
   418→    const line: Box[] = [];
   419→    let rowW = 0;
   420→    const pushRow = () => {
   421→      const total = line.reduce((s, b) => s + b.w + DG_GAP, -DG_GAP);
   422→      let x = Math.max(DG_MARGIN, (DG_W - total) / 2);
   423→      const baseY = y;
   424→      let maxH = 0;
   425→      for (const b of line) {
   426→        b.x = x;
   427→        b.y = baseY;
   428→        x += b.w + DG_GAP;
   429→        maxH = Math.max(maxH, b.h);
   430→        boxes.push(b);
   431→      }
   432→      y = baseY + maxH + ROW_GAP;
   433→    };
   434→    for (const m of metas) {
   435→      const b: Box = { node: m.node, row: ri, x: 0, y: 0, w: m.w, h: m.h, titleLines: m.titleLines };
   436→      if (rowW + m.w > DG_W - 2 * DG_MARGIN && line.length) {
   437→        pushRow();
   438→        line.length = 0;
   439→        rowW = 0;
   440→      }
   441→      line.push(b);
   442→      rowW += m.w + DG_GAP;
   443→    }
   444→    if (line.length) pushRow();
   445→  });
   446→  const vbH = Math.max(y - ROW_GAP + DG_MARGIN, 320);
   447→  const boxById = new Map(boxes.map((b) => [b.node.id, b]));
   448→
   449→  // ---- arrow routing ----
   450→  const edgePath = (a: Box, b: Box) => {
   451→    const xc1 = a.x + a.w / 2;
   452→    const xc2 = b.x + b.w / 2;
   453→    if (b.row > a.row) {
   454→      const y1 = a.y + a.h;
   455→      const y2 = b.y;
   456→      const dy = Math.max(24, Math.abs(y2 - y1) / 2);
   457→      return `M ${xc1} ${y1} C ${xc1} ${y1 + dy}, ${xc2} ${y2 - dy}, ${xc2} ${y2}`;
   458→    }
   459→    if (b.row === a.row) {
   460→      const leftToRight = a.x + a.w <= b.x;
   461→      const rightToLeft = b.x + b.w <= a.x;
   462→      if (leftToRight || rightToLeft) {
   463→        const [s, t] = leftToRight ? [a, b] : [b, a];
   464→        const mx = (s.x + s.w + t.x) / 2;
   465→        return `M ${s.x + s.w} ${s.y + s.h / 2} C ${mx} ${s.y + s.h / 2}, ${mx} ${t.y + t.h / 2}, ${t.x} ${t.y + t.h / 2}`;
   466→      }
   467→      const y = Math.max(a.y + a.h, b.y + b.h);
   468→      return `M ${xc1} ${a.y + a.h} C ${xc1} ${y + 22}, ${xc2} ${y + 22}, ${xc2} ${b.y + b.h}`;
   469→    }
   470→    // backward: loop around the side
   471→    if (a.x >= b.x + b.w) {
   472→      const bx = Math.min(a.x, b.x + b.w) - 44;
   473→      return `M ${a.x} ${a.y + a.h / 2} C ${bx} ${a.y + a.h / 2}, ${bx} ${b.y + b.h / 2}, ${b.x + b.w} ${b.y + b.h / 2}`;
   474→    }
   475→    const bx = Math.max(a.x + a.w, b.x + b.w) + 44;
   476→    return `M ${a.x + a.w} ${a.y + a.h / 2} C ${bx} ${a.y + a.h / 2}, ${bx} ${b.y + b.h / 2}, ${b.x + b.w} ${b.y + b.h / 2}`;
   477→  };
   478→
   479→  // bezier midpoint for the edge label
   480→  const mid = (a: Box, b: Box) => {
   481→    const d = edgePath(a, b);
   482→    const n = d.match(/-?[\d.]+/g)?.map(Number) ?? [];
   483→    if (n.length < 8) return null;
   484→    const [x0, y0, x1, y1, x2, y2, x3, y3] = n;
   485→    return {
   486→      x: (x0 + 3 * x1 + 3 * x2 + x3) / 8,
   487→      y: (y0 + 3 * y1 + 3 * y2 + y3) / 8,
   488→    };
   489→  };
   490→
   491→  const lit = (linkFrom: string, linkTo: string) => !hover || hover === linkFrom || hover === linkTo;
   492→
   493→  return (
   494→    <div className="mt-2">
   495→      {spec.title && <p className="text-sm font-semibold text-stone-900 mb-0.5">{spec.title}</p>}
   496→      {spec.tell && <p className="text-xs text-stone-500 mb-1.5">{spec.tell}</p>}
   497→      <div className="flex flex-wrap items-center gap-x-4 gap-y-1 mb-1.5 text-[10.5px] text-stone-500">
   498→        <span className="flex items-center gap-1"><span className="inline-block w-3 h-3 rounded-sm border" style={{ background: CERT_STYLE.Fact.fill, borderColor: CERT_STYLE.Fact.stroke }} /> proven</span>
   499→        <span className="flex items-center gap-1"><span className="inline-block w-3 h-3 rounded-sm border border-dashed" style={{ background: CERT_STYLE.Inference.fill, borderColor: CERT_STYLE.Inference.stroke }} /> good guess</span>
   500→        <span className="flex items-center gap-1"><span className="inline-block w-3 h-3 rounded-sm border border-dotted" style={{ background: CERT_STYLE.Hypothesis.fill, borderColor: CERT_STYLE.Hypothesis.stroke }} /> not sure yet</span>
   501→      </div>
   502→      <div className="overflow-x-auto rounded-lg border border-stone-200 bg-white">
   503→        <svg
   504→          viewBox={`0 0 ${DG_W} ${vbH}`}
   505→          className="block min-w-[620px]"
   506→          width="100%"
   507→          role="img"
   508→          aria-label={`${spec.title ?? "Diagram"}: ${nodes.length} boxes, ${links.length} connections`}
   509→        >
   510→          <defs>
   511→            <marker id={`${uid}-af`} viewBox="0 0 10 10" refX="9" refY="5" markerWidth="6.5" markerHeight="6.5" orient="auto-start-reverse">
   512→              <path d="M 0 0 L 10 5 L 0 10 z" fill={CERT_STYLE.Fact.stroke} />
   513→            </marker>
   514→            <marker id={`${uid}-ai`} viewBox="0 0 10 10" refX="9" refY="5" markerWidth="6.5" markerHeight="6.5" orient="auto-start-reverse">
   515→              <path d="M 0 0 L 10 5 L 0 10 z" fill={CERT_STYLE.Inference.stroke} />
   516→            </marker>
   517→          </defs>
   518→
   519→          {/* arrows first (under boxes) */}
   520→          {links.map((l, i) => {
   521→            const a = boxById.get(l.from);
   522→            const b = boxById.get(l.to);
   523→            if (!a || !b) return null;
   524→            const fact = (l.certainty ?? "Fact") === "Fact";
   525→            const color = fact ? CERT_STYLE.Fact.stroke : CERT_STYLE.Inference.stroke;
   526→            const on = lit(l.from, l.to);
   527→            const m = mid(a, b);
   528→            return (
   529→              <g key={i} opacity={on ? 0.85 : 0.12} style={{ transition: "opacity 120ms" }}>
   530→                <path
   531→                  d={edgePath(a, b)}
   532→                  fill="none"
   533→                  stroke={color}
   534→                  strokeWidth={on && hover ? 2 : 1.2}
   535→                  strokeDasharray={l.dashed ? "5 4" : undefined}
   536→                  markerEnd={fact ? `url(#${uid}-af)` : `url(#${uid}-ai)`}
   537→                />
   538→                {l.label && m && (!hover || on) && (
   539→                  <text
   540→                    x={m.x}
   541→                    y={m.y - 3}
   542→                    textAnchor="middle"
   543→                    fontSize="8.5"
   544→                    fill="#78716c"
   545→                    stroke="#ffffff"
   546→                    strokeWidth="3"
   547→                    paintOrder="stroke"
   548→                  >
   549→                    {l.label}
   550→                  </text>
   551→                )}
   552→              </g>
   553→            );
   554→          })}
   555→
   556→          {/* boxes — UML look: title compartment + body compartment */}
   557→          {boxes.map((b) => {
   558→            const cert = (b.node.certainty ?? "Inference") as Cert;
   559→            const st = CERT_STYLE[cert];
   560→            const isHover = hover === b.node.id;
   561→            const body = b.node.lines ?? [];
   562→            const bodyLines = body.map((l) => (l.length > 26 ? l.slice(0, 25) + "…" : l));
   563→            const headH = 8 + b.titleLines.length * 13.5;
   564→            const label = `${KIND_PLAIN[b.node.kind] ?? b.node.kind}: ${b.node.title} — ${cert === "Fact" ? "proven" : cert === "Inference" ? "good guess" : "not sure yet"}`;
   565→            return (
   566→              <g
   567→                key={b.node.id}
   568→                tabIndex={0}
   569→                role="button"
   570→                aria-label={label}
   571→                onPointerEnter={() => setHover(b.node.id)}
   572→                onPointerLeave={() => setHover(null)}
   573→                onFocus={() => setHover(b.node.id)}
   574→                onBlur={() => setHover(null)}
   575→                className="outline-none cursor-default"
   576→              >
   577→                <title>{label}</title>
   578→                <rect
   579→                  x={b.x}
   580→                  y={b.y}
   581→                  width={b.w}
   582→                  height={b.h}
   583→                  rx={8}
   584→                  fill={st.fill}
   585→                  stroke={isHover ? "#1c1917" : st.stroke}
   586→                  strokeWidth={isHover ? 2.4 : 1.4}
   587→                  strokeDasharray={st.dash}
   588→                  style={{ transition: "stroke-width 100ms" }}
   589→                />
   590→                {/* title compartment */}
   591→                <line x1={b.x} y1={b.y + headH} x2={b.x + b.w} y2={b.y + headH} stroke={st.stroke} strokeWidth={0.8} opacity={0.55} />
   592→                {b.titleLines.map((t, ti) => (
   593→                  <text
   594→                    key={ti}
   595→                    x={b.x + b.w / 2}
   596→                    y={b.y + 17 + ti * 13.5}
   597→                    textAnchor="middle"
   598→                    fontSize={b.titleLines.length > 1 ? 10.5 : 11.5}
   599→                    fontWeight={700}
   600→                    fill={st.text}
   601→                  >
   602→                    {t}
   603→                  </text>
   604→                ))}
   605→                {/* body compartment */}
   606→                {bodyLines.map((t, li) => (
   607→                  <text key={li} x={b.x + 10} y={b.y + headH + 14 + li * 12.5} fontSize={10} fill="#57534e">
   608→                    {t}
   609→                  </text>
   610→                ))}
   611→                {b.node.more ? (
   612→                  <text x={b.x + 10} y={b.y + headH + 14 + bodyLines.length * 12.5} fontSize={9} fontStyle="italic" fill="#a8a29e">
   613→                    …and {b.node.more} more
   614→                  </text>
   615→                ) : null}
   616→              </g>
   617→            );
   618→          })}
   619→        </svg>
   620→      </div>
   621→      {nodes.length >= 12 && (
   622→        <p className="text-[10px] text-stone-400 mt-1 flex items-center gap-1">
   623→          <MoveVertical className="h-3 w-3" /> scroll sideways if the picture is wider than this box — or press “Show this chart big on the canvas”
   624→        </p>
   625→      )}
   626→    </div>
   627→  );
   628→}

   638→}
