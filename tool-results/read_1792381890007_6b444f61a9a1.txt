    19→    19→function cut(s: string, n: number): string {
    20→    20→  return s.length > n ? s.slice(0, n - 1) + "…" : s;

    23→    23→function spec(kind: ChatChartSpec["kind"], title: string, tell: string, source?: string): ChatChartSpec {
    24→    24→  return { kind, engine: engineFor(kind), layout: layoutFor(kind), title, tell, source };

    31→    31→export function buildConceptTreeSpec(concepts: ConceptTree | null): ChatChartSpec | null {
    32→    32→  if (!concepts?.nodes?.length) return null;
    33→    33→  const kinds = new Set(concepts.nodes.map((n) => n.kind));
    34→    34→  if (!kinds.has("Domain")) return null; // no hierarchy to draw — honest empty
    35→    35→
    36→    36→  const tree: TreeNode[] = concepts.nodes
    37→    37→    .slice(0, 40)
    38→    38→    .map((n) => ({
    39→    39→      id: `c${n.id}`,
    40→    40→      label: cut(n.label, 30),
    41→    41→      kind: n.kind,
    42→    42→      parent: n.parent === null || n.parent === undefined ? null : `c${n.parent}`,
    43→    43→      note: n.behaviors?.length
    44→    44→        ? `${n.behaviors.length} ${n.behaviors.length === 1 ? "behaviour" : "behaviours"}`
    45→    45→        : undefined,
    46→    46→      certainty: undefined,
    47→    47→    }));
    48→    48→
    49→    49→  const roots = tree.filter((t) => t.parent === null).length;
    50→    50→  return {
    51→    51→    ...spec("tree", "What the project is made of", `The idea hierarchy: ${roots} big-picture ${roots === 1 ? "idea" : "ideas"}, the files under them, and what each does. Tap any node to ask about it.`),
    52→    52→    tree,
    53→    53→    source: "concepts.json (the engine's concept tree)",
    54→    54→  };
    55→    55→}

    61→    61→export function buildCoverageHeatmapSpec(quality: CieQuality | null): ChatChartSpec | null {
    62→    62→  const cov = quality?.coverage_heatmap;
    63→    63→  if (!cov?.files?.length) return null;

    67→    67→    label: f.covered != null && f.statements != null ? `${f.covered} of ${f.statements} lines` : "never even imported",
    68→    68→  }));
    69→    69→  const pct = cov.totals?.percent;
    70→    70→  return {
    71→    71→    ...spec(
    72→    72→      "heatmap",
    73→    73→      "Test coverage, file by file (measured)",
    74→    74→      `Measured by ${cov.tool ?? "coverage.py"} on the sample's real test run. ${pct != null ? `Whole project: ${pct}% of statements.` : ""} Green is well covered; rose was never even imported — the label says so.`,
    75→    75→    ),
    76→    76→    tiles,
    77→    77→    source: "quality.json — coverage_heatmap (coverage.py)",
    78→    78→  };
    79→    79→}

   366→   366→function fileFacts(q: CieQuality, name: string): QualityAnswer | null {
   367→   367→  // exact filename tokens first ("auth.py", "001_init.sql"), then stem match
   368→   368→  const tokens: string[] = name.toLowerCase().match(/[a-z0-9_\-]+\.(py|sql|md|txt|sh)/g) ?? [];
   369→   369→  const exactFile = tokens.length
   370→   370→    ? q.complexity.files.find((f) => tokens.includes(f.file.toLowerCase()))?.file ??
   371→   371→      q.coverage_heatmap.files.find((f) => tokens.includes(f.file.toLowerCase()))?.file
   372→   372→    : undefined;
   373→   373→
   374→   374→  const base = (exactFile ?? name).toLowerCase().replace(/\.(py|sql|md|txt|sh)$/, "");
   375→   375→  const matchFile = (f: string) => {
   376→   376→    const lf = f.toLowerCase();
   377→   377→    if (exactFile) return lf === exactFile.toLowerCase();
   378→   378→    return lf.includes(base) && base.length > 2;
   379→   379→  };
   380→   380→  if (base.length <= 2 && !exactFile) return null;
   381→   381→
   382→   382→  const cc = q.complexity.files.find((f) => matchFile(f.file));
   383→   383→  const churn = q.churn.sample.per_file.find((r) => matchFile(r.file));
   384→   384→  const sec = q.security.issues.filter((it) => matchFile(it.file));
   385→   385→  const cov = q.coverage_heatmap.files.find((f) => matchFile(f.file));
   386→   386→  const spot = q.hotspots.rows.find((r) => matchFile(r.file));
   387→   387→  const drift = q.drift.code.find((c) => matchFile(c.file));
   388→   388→  if (!cc && !churn && !sec.length && !cov && !spot) return null;
   389→   389→
   390→   390→  const lines: string[] = [];
   391→   391→  lines.push(`**Everything collected about \`${name}\`**`);
   392→   392→  if (cc) lines.push(`**Complexity:** ${cc.functions} functions, avg CC ${cc.avg_cc}, max ${cc.max_cc} (formula: ${q.complexity.formula}).`);
   393→   393→  if (churn) lines.push(`**Churn:** ${churn.added} lines added + ${churn.deleted} deleted over ${churn.commits} commit(s); last changed ${churn.last_change ?? "unknown"}.`);
   394→   394→  if (cov) lines.push(`**Test coverage:** ${cov.percent}% (${cov.note ?? (cov.covered != null ? `${cov.covered} of ${cov.statements} lines` : "never even imported")}) — measured by ${q.coverage_heatmap.tool}.`);
   395→   395→  if (spot) lines.push(`**Hot-spot risk:** ${spot.risk}/100 (${q.hotspots.formula}).`);
   396→   396→  if (sec.length) {
   397→   397→    lines.push(`**Security findings:** ${sec.map((it) => `${it.name} (${it.cwe}) at line ${it.line}`).join("; ")}.`);
   398→   398→  } else if (cc || cov) {
   399→   399→    lines.push(`**Security findings:** none on this file from the disclosed rule set.`);
   400→   400→  }
   401→   401→  if (drift) lines.push(`**Drift:** ${drift.note}; code last changed ${drift.last_change}, tests ${drift.tests_last_change ?? "none"}.`);
   402→   402→  lines.push(`_From quality.json — collected by the ETL collector on ${q.collected_at.slice(0, 10)}._`);
   403→   403→  return {
   404→   404→    answer: lines.join("\n\n"),
   405→   405→    chart: buildCoverageHeatmapSpec(q),
   406→   406→  };
   407→   407→}

   409→   409→function securityFacts(q: CieQuality, needle: string): QualityAnswer | null {
   410→   410→  const n = needle.toLowerCase();
   411→   411→  // a generic "are there security problems" question should NOT pin one
   412→   412→  // category — only match when a concrete kind / CWE / rule is named
   413→   413→  const hitCategory = Object.keys(q.security.by_category ?? {}).find((c) => n.includes(c.toLowerCase().slice(0, 14)));
   414→   414→  const hitIssue = q.security.issues.find((it) => n.includes(it.cwe.toLowerCase()) || n.includes(it.rule.toLowerCase()) || n.includes(it.name.toLowerCase()));
   415→   415→  if (!hitCategory && !hitIssue) return null;
   416→   416→  const name = hitIssue?.name ?? hitCategory!;
   417→   417→  const rows = q.security.issues.filter((it) => it.name === name || (hitIssue && it.id === hitIssue.id));
   418→   418→  const list = rows.map((it) => `- ${it.file}:${it.line} — ${it.cwe}, ${it.severity.toLowerCase()} urgency. Why it matters: ${it.why}`);
   419→   419→  const answer = [
   420→   420→    `**${name} — ${rows.length} finding(s) in the sample**`,
   421→   421→    list.join("\n"),
   422→   422→    `The scan is ${q.security.tool} — heuristics with file:line evidence, not a full SAST scanner. On production projects the same slots come from ${q.security.real_world_equivalents?.join(", ") ?? "bandit, semgrep, CodeQL"}.`,
   423→   423→    `_Ask about a specific file name to see its full collected facts._`,
   424→   424→  ].join("\n\n");
   425→   425→  const data = Object.entries(q.security.by_category ?? {}).map(([k, v]) => ({
   426→   426→    name: cut(k, 24),
   427→   427→    value: v,
   428→   428→    color: k.includes("Hardcoded") || k.includes("SQL") ? "rose" : "amber",
   429→   429→  }));
   430→   430→  return {
   431→   431→    answer,
   432→   432→    chart: data.length
   433→   433→      ? { ...spec("hbar", "Security issues by kind", "Counts per finding kind, from the disclosed rule set."), data, source: "quality.json — security scan" }
   434→   434→      : null,
   435→   435→  };
   436→   436→}

   438→   438→function functionFacts(q: CieQuality, needle: string): QualityAnswer | null {
   439→   439→  const n = needle.toLowerCase();
   440→   440→  const fn = q.complexity.functions.find((f) => f.function.toLowerCase() === n.replace(/[^a-z0-9_]/g, "") || n.includes(` ${f.function.toLowerCase()} `) || n.includes(`"${f.function.toLowerCase()}"`));
   441→   441→  if (!fn) return null;
   442→   442→  const churnRow = q.churn.sample.per_file.find((r) => r.file === fn.file);
   443→   443→  const cov = q.coverage_heatmap.files.find((f) => f.file === fn.file);
   444→   444→  const answer = [
   445→   445→    `**\`${fn.function}\` (in ${fn.file}, line ${fn.line})**`,
   446→   446→    `Cyclomatic complexity **${fn.cc}** — ${q.complexity.formula}. Its file's max CC is ${q.complexity.files.find((f) => f.file === fn.file)?.max_cc}.`,
   447→   447→    churnRow ? `The file churned ${churnRow.added + churnRow.deleted} lines over ${churnRow.commits} commit(s).` : "",
   448→   448→    cov ? `Test coverage of ${fn.file}: ${cov.percent}%.` : "",
   449→   449→    `Measured by ${q.complexity.tool}. On production projects: ${q.complexity.real_world_equivalents?.join(", ") ?? "radon, lizard, SonarQube"}.`,
   450→   450→  ].filter(Boolean).join("\n\n");
   451→   451→  return { answer, chart: null };
   452→   452→}

   454→   454→function metricFacts(q: CieQuality, needle: string): QualityAnswer | null {
   455→   455→  const n = needle.toLowerCase();
   456→   456→  const m = q.metrics.find((mm) =>
   457→   457→    (mm.id === "defect-density" && (n.includes("defect density") || n.includes("density"))) ||
   458→   458→    (mm.id === "mttf-mtbf" && (n.includes("mtbf") || n.includes("mttf") || n.includes("between failures"))) ||
   459→   459→    (mm.id === "test-coverage" && n.includes("coverage") && !n.includes("heatmap") && !n.includes("heat map")) ||
   460→   460→    (mm.id === "cyclomatic" && (n.includes("cyclomatic") || (n.includes("complex") && n.includes("control")))) ||
   461→   461→    (mm.id === "arrival-rate" && (n.includes("arrival") || n.includes("how fast are defects"))) ||
   462→   462→    (mm.id === "code-churn" && n.includes("churn")) ||
   463→   463→    (mm.id === "crash-rate" && (n.includes("crash") || n.includes("hang"))) ||
   464→   464→    (mm.id === "mttr" && (n.includes("mttr") || n.includes("repair") || n.includes("time to resolve")))
   465→   465→  );
   466→   466→  if (!m) return null;
   467→   467→  const answer = [
   468→   468→    `**${m.name}: ${m.value_text}**`,
   469→   469→    `**What it measures:** ${m.what_it_measures}. **Typical calculation:** ${m.typical_calculation}.`,
   470→   470→    `**How it was computed here:** ${m.how_computed_here}`,
   471→   471→    `**Source:** ${m.source} · **Scope:** ${m.scope}.`,
   472→   472→    q.defects.derived.proxy_note && (m.id === "defect-density" || m.id === "mttf-mtbf" || m.id === "arrival-rate") ? `**Proxy disclosure:** ${q.defects.derived.proxy_note}` : "",
   473→   473→  ].filter(Boolean).join("\n\n");
   474→   474→  return { answer, chart: null };
   475→   475→}

   477→   477→export function tryAnswerQuality(question: string, quality: CieQuality | null, concepts: ConceptTree | null): QualityAnswer | null {
   478→   478→  if (!quality) return null;
   479→   479→  const q = quality;
   480→   480→  const n = question.toLowerCase();
   481→   481→
   482→   482→  // ---------- element-level answers FIRST (a tapped bar/tile/row must win
   483→   483→  // over the whole-chart answer, whose keywords often appear in the same
   484→   484→  // question) ----------
   485→   485→  const fileAns = fileFacts(q, question);
   486→   486→  if (fileAns) return fileAns;
   487→   487→  const fnAns = functionFacts(q, question);
   488→   488→  if (fnAns) return fnAns;
   489→   489→  const secEl = securityFacts(q, question);
   490→   490→  if (secEl) return secEl;
   491→   491→  const dayMatch = question.match(/\d{4}-\d{2}-\d{2}/);
   492→   492→  if (dayMatch) {
   493→   493→    const row = q.churn.hosting_repo.daily.find((d) => d.day === dayMatch[0]);
   494→   494→    if (row) {
   495→   495→      return {
   496→   496→        answer: `**Churn on ${row.day}** — ${row.added} lines added, ${row.deleted} deleted (${row.churn} total), from ${q.churn.tool} over the hosting repository's real history. The biggest day of this sprint is the day the sample playground landed.`,
   497→   497→        chart: null,
   498→   498→      };
   499→   499→    }
   500→   500→  }
   501→   501→
   502→   502→  // ---------- whole-chart keyword answers ----------
   503→   503→  const wants = (w: string[]) => w.some((k) => n.includes(k));
   504→   504→
   505→   505→  if (wants(["hierarchy", "made of"]) && !wants(["file "])) {
   506→   506→    const s = buildConceptTreeSpec(concepts);
   507→   507→    if (s) return { answer: `**The concept hierarchy** — the big picture first, then the files under it, then what each does. Every node comes from the engine's concept tree; tap any node to ask about it.\n\n_Source: concepts.json._`, chart: s };
   508→   508→  }
   509→   509→  if (wants(["security", "insecure", "secret", "credential", "vulnerab", "weakness"])) {
   510→   510→    const sec = securityFacts(q, question) ?? fileFacts(q, question);
   511→   511→    if (sec) return sec;
   512→   512→    const data = Object.entries(q.security.by_category ?? {}).map(([k, v]) => ({ name: cut(k, 24), value: v, color: "amber" as const }));
   513→   513→    if (data.length) {
   514→   514→      return {
   515→   515→        answer: `**The security scan found ${q.security.issues.length} findings** (${Object.entries(q.security.by_severity ?? {}).map(([s, c]) => `${c} ${s.toLowerCase()}`).join(", ")}). Kinds: ${Object.entries(q.security.by_category).map(([k, v]) => `${v}× ${k}`).join(", ")}.\n\nThe scan is ${q.security.tool} — ${q.security.honesty} On production projects: ${q.security.real_world_equivalents?.join(", ")}.`,
   516→   516→        chart: { ...spec("hbar", "Security issues by kind", "Counts per finding kind."), data, source: "quality.json — security scan" },
   517→   517→      };
   518→   518→    }
   519→   519→  }
   520→   520→  if (wants(["duplicate", "twice", "copy-paste", "copy paste"])) {
   521→   521→    const dup = q.duplication;
   522→   522→    const rows = (dup.pairs ?? []).slice(0, 6).map((p) => `- ${p.a_file}:${p.a_line} ↔ ${p.b_file}:${p.b_line} — ${p.lines} lines (${p.preview || "same block"})`);
   523→   523→    return {
   524→   524→      answer: rows.length
   525→   525→        ? `**${dup.blocks_found} duplicated block(s), ${dup.duplicated_lines} lines** — matched by ${dup.tool}:\n${rows.join("\n")}\n\nHere the duplicated blocks are the CREATE TABLE definitions shared between the migrations and the schema dump — a classic sample of schema defined twice.`
   526→   526→        : `No repeated blocks found by ${dup.tool} — recorded as found.`,
   527→   527→      chart: null,
   528→   528→    };
   529→   529→  }
   530→   530→  if (wants(["hot spot", "hotspot", "hot spot"])) {
   531→   531→    const rows = (q.hotspots.rows ?? []).slice(0, 5).map((r) => `- ${r.file} — risk ${r.risk}/100 (churn ${r.churn}, max CC ${r.max_cc}, urgent findings ${r.urgent_findings})`);
   532→   532→    return { answer: `**Hot spots, ranked** by the disclosed formula: ${q.hotspots.formula}.\n${rows.join("\n")}\n\nOn production projects: ${q.hotspots.real_world_equivalents?.join(", ") ?? "CodeScene, SonarQube"}.`, chart: null };
   533→   533→  }
   534→   534→  if (wants(["heatmap", "heat map", "coverage map"]) || (wants(["coverage"]) && wants(["every", "file by file", "grid"]))) {
   535→   535→    const s = buildCoverageHeatmapSpec(q);
   536→   536→    if (s) return { answer: `**The coverage heat map** — every file, measured by ${q.coverage_heatmap.tool}. ${q.coverage_heatmap.honesty}\n\n_Tap any tile to ask about that file._`, chart: s };
   537→   537→  }
   538→   538→  if (wants(["code drift", "ahead of its tests", "test lag"])) {
   539→   539→    const rows = (q.drift.code ?? []).slice(0, 6).map((c) => `- ${c.file} — code ${c.last_change}, tests ${c.tests_last_change ?? "none"}${c.test_lag_days != null ? ` (lag ${c.test_lag_days}d)` : ""}`);
   540→   540→    return { answer: `**Code drift vs tests** (${q.drift.tool}):\n${rows.join("\n")}\n\n${q.drift.honesty}`, chart: null };
   541→   541→  }
   542→   542→  if (wants(["document drift", "docs behind", "words fell"])) {
   543→   543→    const rows = (q.drift.document ?? []).slice(0, 6).map((d) => `- ${d.doc} — ${d.note}${d.drift_days > 0 ? ` (${d.drift_days}d behind)` : ""}`);
   544→   544→    return { answer: `**Document drift** (${q.drift.tool}):\n${rows.join("\n")}\n\n${q.drift.honesty}`, chart: null };
   545→   545→  }
   546→   546→
   547→   547→  // metric answers
   548→   548→  const mAns = metricFacts(q, question);
   549→   549→  if (mAns) return mAns;
   550→   550→
   551→   551→  return null;
   552→   552→}
