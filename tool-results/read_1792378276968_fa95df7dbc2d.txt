     1→//! Deep Python analyzer built on tree-sitter.
     2→//!
     3→//! Extracts the full fact set: symbols (modules/classes/functions/methods),
     4→//! calls, returns (with constant normalization), exceptions and handler
     5→//! bodies, branches, loops (with body ranges for N+1), assignments (for
     6→//! taint propagation), taint sources/sinks, string literals, imports and
     7→//! async facts. Unmodeled constructs are recorded as `unsupported`.
     8→

    90→pub fn extract_python(path: &str, src: &str) -> LangFacts {
    91→    let mut parser = Parser::new();
    92→    if parser.set_language(&to_language()).is_err() {
    93→        let mut f = LangFacts::new(path, LangId::Python, "tree-sitter-deep:python");
    94→        f.unsupported.push("parser_unavailable".into());
    95→        return f;
    96→    }
    97→    let Some(tree) = parser.parse(src, None) else {
    98→        let mut f = LangFacts::new(path, LangId::Python, "tree-sitter-deep:python");
    99→        f.unsupported.push("parse_error".into());
   100→        return f;
   101→    };
   102→
   103→    let bytes = src.as_bytes();
   104→    let module_name = cie_adapters::stem_module(path);
   105→    let mut facts = LangFacts::new(path, LangId::Python, "tree-sitter-deep:python");
   106→
   107→    // Local symbol 0 = the module.
   108→    facts.symbols.push(SymbolFact {
   109→        id: SymbolId(0),
   110→        kind: SymbolKind::Module,
   111→        name: module_name.clone(),
   112→        qual_name: module_name.clone(),
   113→        loc: SourceLocation::span_bytes(FileId(0), 0, tree.root_node().end_byte() as u32),
   114→        lang: LangId::Python,
   115→        parent: None,
   116→        rev: 0,
   117→        doc: None,
   118→    });
   119→
   120→    let mut ctx = Ctx {
   121→        path,
   122→        bytes,
   123→        module: SymbolId(0),
   124→        class: None,
   125→        func: SymbolId(0),
   126→        loop_stack: vec![],
   127→        handler_depth: 0,
   128→        next_id: SymbolId(1),
   129→    };
   130→
   131→    let root = tree.root_node();
   132→    walk(&mut facts, &mut ctx, root);
   133→    facts
   134→}
   135→
   136→struct Ctx<'a> {
   137→    path: &'a str,
   138→    bytes: &'a [u8],
   139→    module: SymbolId,
   140→    class: Option<SymbolId>,
   141→    func: SymbolId,
   142→    loop_stack: Vec<(u32, u32)>,
   143→    handler_depth: u32,
   144→    next_id: SymbolId,
   145→}

   147→fn text<'a>(node: Node, bytes: &'a [u8]) -> &'a str {
   148→    node.utf8_text(bytes).unwrap_or("")
   149→}
   150→
   151→fn line_of(node: Node) -> u32 {
   152→    node.start_position().row as u32 + 1
   153→}
   154→
   155→/// §14.1 byte span of a node: `(start_byte, end_byte)` straight from
   156→/// tree-sitter (TASK-006).
   157→fn span_of(node: Node) -> (u32, u32) {
   158→    let r = node.byte_range();
   159→    (r.start as u32, r.end as u32)
   160→}

   162→/// Normalize a return expression into a constant class.
   163→/// "true"/"false"/"none"/"num"/"str"/"empty" — or None when not constant.
   164→pub fn normalize_constant(expr: &str) -> Option<String> {
   165→    let t = expr.trim();
   166→    if t.is_empty() {
   167→        return None;
   168→    }
   169→    match t {
   170→        "True" | "true" => Some("true".into()),
   171→        "False" | "false" => Some("false".into()),
   172→        "None" | "null" | "nil" | "undefined" => Some("none".into()),
   173→        "{}" | "[]" | "()" | "{,}" => Some("empty".into()),
   174→        _ => {
   175→            if t.parse::<f64>().is_ok() {
   176→                Some("num".into())
   177→            } else if (t.starts_with('"') && t.ends_with('"'))
   178→                || (t.starts_with('\'') && t.ends_with('\''))
   179→            {
   180→                if t.len() == 2 {
   181→                    Some("empty".into())
   182→                } else {
   183→                    Some("str".into())
   184→                }
   185→            } else if t.starts_with("f\"") || t.starts_with("f'") {
   186→                Some("fstr".into())
   187→            } else {
   188→                None
   189→            }
   190→        }
   191→    }
   192→}
   193→
   194→pub fn is_success_const(c: &str) -> bool {
   195→    matches!(c, "true" | "num" | "str" | "empty")
   196→}

   198→fn push_symbol(
   199→    facts: &mut LangFacts,
   200→    ctx: &mut Ctx,
   201→    kind: SymbolKind,
   202→    name: &str,
   203→    node: Node,
   204→    doc: Option<String>,
   205→) -> SymbolId {
   206→    let id = ctx.next_id;
   207→    ctx.next_id = SymbolId(id.0 + 1);
   208→    let qual = match (kind, ctx.class) {
   209→        (SymbolKind::Method, Some(cls)) => {
   210→            format!(
   211→                "{}.{}.{}",
   212→                facts.symbols[0].name, facts.symbols[cls.0 as usize].name, name
   213→            )
   214→        }
   215→        (SymbolKind::Method, None) => {
   216→            format!("{}.{}", facts.symbols[0].name, name)
   217→        }
   218→        _ => {
   219→            if let Some(class_id) = ctx.class {
   220→                format!(
   221→                    "{}.{}.{}",
   222→                    facts.symbols[0].name,
   223→                    facts.symbols[class_id.0 as usize].name,
   224→                    name
   225→                )
   226→            } else {
   227→                format!("{}.{}", facts.symbols[0].name, name)
   228→            }
   229→        }
   230→    };
   231→    let (sl, be) = span_of(node);
   232→    facts.symbols.push(SymbolFact {
   233→        id,
   234→        kind,
   235→        name: name.to_string(),
   236→        qual_name: qual,
   237→        loc: SourceLocation::span_bytes(FileId(0), sl, be),
   238→        lang: LangId::Python,
   239→        parent: match kind {
   240→            SymbolKind::Method => ctx.class,
   241→            _ => Some(ctx.module),
   242→        },
   243→        rev: 0,
   244→        doc,
   245→    });
   246→    id
   247→}
   248→
   249→fn docstring_of(body: Node, bytes: &[u8]) -> Option<String> {
   250→    let mut c = body.walk();
   251→    let first = body.children(&mut c).next()?;
   252→    if first.kind() == "expression_statement" {
   253→        let mut c2 = first.walk();
   254→        let inner = first.children(&mut c2).next()?;
   255→        if inner.kind() == "string" {
   256→            let t = text(inner, bytes);
   257→            let t = t.trim_matches(|ch| ch == '"' || ch == '\'');
   258→            let t = t.trim();
   259→            if !t.is_empty() {
   260→                return Some(t.chars().take(400).collect());
   261→            }
   262→        }
   263→    }
   264→    None
   265→}

   267→fn walk(facts: &mut LangFacts, ctx: &mut Ctx, node: Node) {
   268→    let kind = node.kind();
   269→
   270→    match kind {
   271→        "function_definition" => {
   272→            let is_async = node.children(&mut node.walk()).any(|c| c.kind() == "async");
   273→            let name_node = node.child_by_field_name("name").unwrap_or(node);
   274→            let name = text(name_node, ctx.bytes).to_string();
   275→            let s_kind = if ctx.class.is_some() {
   276→                if name.starts_with("test_") {
   277→                    SymbolKind::Test
   278→                } else {
   279→                    SymbolKind::Method
   280→                }
   281→            } else if name.starts_with("test_") {
   282→                SymbolKind::Test
   283→            } else {
   284→                SymbolKind::Function
   285→            };
   286→            let id = push_symbol(facts, ctx, s_kind, &name, node, None);
   287→            if is_async {
   288→                facts.async_facts.push(AsyncSite {
   289→                    symbol: id,
   290→                    kind: AsyncKind::AsyncFn,
   291→                    line: line_of(node),
   292→                });
   293→            }
   294→            if let Some(body) = node.child_by_field_name("body") {
   295→                let doc = docstring_of(body, ctx.bytes);
   296→                if doc.is_some() {
   297→                    facts.symbols[id.0 as usize].doc = doc;
   298→                }
   299→                let prev_fn = ctx.func;
   300→                ctx.func = id;
   301→                walk_children(facts, ctx, body);
   302→                ctx.func = prev_fn;
   303→            }
   304→        }
   305→        "class_definition" => {
   306→            let name_node = node.child_by_field_name("name").unwrap_or(node);
   307→            let name = text(name_node, ctx.bytes).to_string();
   308→            let id = push_symbol(facts, ctx, SymbolKind::Class, &name, node, None);
   309→            let prev_class = ctx.class;
   310→            ctx.class = Some(id);
   311→            if let Some(body) = node.child_by_field_name("body") {
   312→                walk_children(facts, ctx, body);
   313→            }
   314→            ctx.class = prev_class;
   315→        }
   316→        "decorated_definition" => {
   317→            let mut c = node.walk();
   318→            for child in node.children(&mut c) {
   319→                match child.kind() {
   320→                    "decorator" => {
   321→                        // Record decorator as a call (e.g. @app.route(...)).
   322→                        let t = text(child, ctx.bytes).trim_start_matches('@').to_string();
   323→                        if !t.is_empty() {
   324→                            facts.calls.push(CallSite {
   325→                                caller: ctx.func,
   326→                                callee_text: t,
   327→                                callee: Callee::Unresolved("pending".into()),
   328→                                line: line_of(child),
   329→                                in_loop: false,
   330→                                arg_vars: vec![],
   331→                            });
   332→                        }
   333→                    }
   334→                    _ => walk(facts, ctx, child),
   335→                }
   336→            }
   337→        }
   338→        "call" => {
   339→            let fn_node = node.child_by_field_name("function").unwrap_or(node);
   340→            let callee_text = text(fn_node, ctx.bytes).to_string();
   341→            let line = line_of(node);
   342→            let (cb, ce) = span_of(node);
   343→            let in_loop = ctx
   344→                .loop_stack
   345→                .iter()
   346→                // loop_stack entries are §14.1 byte spans (TASK-006)
   347→                .any(|(s, e)| cb >= *s && ce <= *e);
   348→
   349→            // Argument variables + whole-args text (for sink/source checks).
   350→            let mut arg_vars = vec![];
   351→            let mut args_text = String::new();
   352→            if let Some(args) = node.child_by_field_name("arguments") {
   353→                args_text = text(args, ctx.bytes).to_string();
   354→                collect_identifiers(args, ctx.bytes, &mut arg_vars);
   355→            }
   356→
   357→            let callee = if let Some(b) = effects::classify_builtin(&callee_text) {
   358→                Callee::Builtin(b)
   359→            } else {
   360→                Callee::Unresolved("pending".into())
   361→            };
   362→            facts.calls.push(CallSite {
   363→                caller: ctx.func,
   364→                callee_text,
   365→                callee,
   366→                line,
   367→                in_loop,
   368→                arg_vars: dedup(arg_vars),
   369→            });
   370→
   371→            // Sink detection.
   372→            if let Some(sink_kind) =
   373→                effects::classify_sink(&facts.calls.last().unwrap().callee_text, &args_text)
   374→            {
   375→                let parameterized = args_text.contains('?')
   376→                    || args_text.contains("%s")
   377→                    || args_text.contains("%(")
   378→                    || args_text.contains("${");
   379→                facts.sinks.push(SinkSite {
   380→                    symbol: ctx.func,
   381→                    call_text: facts.calls.last().unwrap().callee_text.clone(),
   382→                    arg_vars: facts.calls.last().unwrap().arg_vars.clone(),
   383→                    line,
   384→                    kind: sink_kind,
   385→                    parameterized,
   386→                });
   387→            }
   388→
   389→            // Recurse into arguments for nested calls.
   390→            if let Some(args) = node.child_by_field_name("arguments") {
   391→                walk_children(facts, ctx, args);
   392→            }
   393→        }
   394→        "return_statement" => {
   395→            let mut parts = vec![];
   396→            let mut c = node.walk();
   397→            for child in node.children(&mut c) {
   398→                if child.kind() != "return" {
   399→                    parts.push(text(child, ctx.bytes).to_string());
   400→                }
   401→            }
   402→            let expr = parts.join(" ").trim().to_string();
   403→            let constant = normalize_constant(&expr);
   404→            facts.returns.push(ReturnSite {
   405→                symbol: ctx.func,
   406→                line: line_of(node),
   407→                expr,
   408→                constant,
   409→            });
   410→            let mut c = node.walk();
   411→            for child in node.children(&mut c) {
   412→                walk(facts, ctx, child);
   413→            }
   414→        }
   415→        "raise_statement" => {
   416→            let t = text(node, ctx.bytes)
   417→                .trim_start_matches("raise ")
   418→                .to_string();
   419→            let kind_text = t
   420→                .split(['(', ' ', '\n'])
   421→                .next()
   422→                .unwrap_or("")
   423→                .to_string();
   424→            facts.exceptions.push(ExceptionSite {
   425→                symbol: ctx.func,
   426→                kind_text,
   427→                is_bare: false,
   428→                line: line_of(node),
   429→                in_handler: ctx.handler_depth > 0,
   430→                handler_end: 0,
   431→            });
   432→        }
   433→        "try_statement" => {
   434→            let mut c = node.walk();
   435→            for child in node.children(&mut c) {
   436→                match child.kind() {
   437→                    "except_clause" => {
   438→                        // type child may be absent -> bare except.
   439→                        // Children between the `except` keyword and `as`/block:
   440→                        let mut type_text = String::new();
   441→                        let mut after_except = false;
   442→                        let mut cc = child.walk();
   443→                        let children: Vec<Node> = child.children(&mut cc).collect();
   444→                        for gc in children {
   445→                            match gc.kind() {
   446→                                "except" => after_except = true,
   447→                                "as" | "block" | "comment" => after_except = false,
   448→                                k if after_except
   449→                                    && matches!(
   450→                                        k,
   451→                                        "identifier"
   452→                                            | "attribute"
   453→                                            | "tuple"
   454→                                            | "string"
   455→                                            | "dotted_name"
   456→                                    ) =>
   457→                                {
   458→                                    type_text = text(gc, ctx.bytes).to_string();
   459→                                    after_except = false;
   460→                                }
   461→                                _ => {}
   462→                            }
   463→                        }
   464→                        let handler_end = child
   465→                            .children(&mut child.walk())
   466→                            .filter(|c| c.kind() == "block")
   467→                            .last()
   468→                            .map(|b| b.end_position().row as u32 + 1)
   469→                            .unwrap_or(line_of(child));
   470→                        facts.exceptions.push(ExceptionSite {
   471→                            symbol: ctx.func,
   472→                            is_bare: type_text.is_empty(),
   473→                            kind_text: type_text,
   474→                            line: line_of(child),
   475→                            in_handler: false,
   476→                            handler_end,
   477→                        });
   478→                        if let Some(block) = child
   479→                            .children(&mut child.walk())
   480→                            .filter(|c| c.kind() == "block")
   481→                            .last()
   482→                        {
   483→                            let prev = ctx.handler_depth;
   484→                            ctx.handler_depth += 1;
   485→                            walk_children(facts, ctx, block);
   486→                            ctx.handler_depth = prev;
   487→                        }
   488→                    }
   489→                    "else_clause" | "finally_clause" => walk_children(facts, ctx, child),
   490→                    "block" => walk_children(facts, ctx, child),
   491→                    _ => {}
   492→                }
   493→            }
   494→        }
   495→        "if_statement" | "elif_clause" => {
   496→            if let Some(cond) = node.child_by_field_name("condition") {
   497→                let cond_text = text(cond, ctx.bytes).to_string();
   498→                let mut vars = vec![];
   499→                collect_identifiers(cond, ctx.bytes, &mut vars);
   500→                let (isl, ibe) = span_of(node);
   501→                facts.ifs.push(IfSite {
   502→                    symbol: ctx.func,
   503→                    line: line_of(node),
   504→                    cond: cond_text,
   505→                    cond_vars: dedup(vars),
   506→                    loc: SourceLocation::span_bytes(FileId(0), isl, ibe),
   507→                });
   508→            }
   509→            let mut c = node.walk();
   510→            for child in node.children(&mut c) {
   511→                walk(facts, ctx, child);
   512→            }
   513→        }
   514→        "for_statement" | "while_statement" => {
   515→            let (bs, be) = node
   516→                .child_by_field_name("body")
   517→                .map(span_of)
   518→                .unwrap_or((line_of(node), line_of(node)));
   519→            let (lsl, lbe) = span_of(node);
   520→            facts.loops.push(LoopSite {
   521→                symbol: ctx.func,
   522→                kind: if kind == "for_statement" {
   523→                    LoopKind::For
   524→                } else {
   525→                    LoopKind::While
   526→                },
   527→                line: line_of(node),
   528→                body_start: bs,
   529→                body_end: be,
   530→                loc: SourceLocation::span_bytes(FileId(0), lsl, lbe),
   531→            });
   532→            ctx.loop_stack.push((bs, be));
   533→            walk_children(facts, ctx, node);
   534→            ctx.loop_stack.pop();
   535→        }
   536→        "assignment" | "augmented_assignment" => {
   537→            let left = node.child_by_field_name("left");
   538→            let right = node.child_by_field_name("right");
   539→            if let (Some(left), Some(right)) = (left, right) {
   540→                if left.kind() == "identifier" {
   541→                    let var = text(left, ctx.bytes).to_string();
   542→                    let rhs_text = text(right, ctx.bytes).to_string();
   543→                    let (rhs_kind, rhs_data) = classify_rhs(right, ctx.bytes, &rhs_text);
   544→                    facts.assignments.push(AssignmentFact {
   545→                        symbol: ctx.func,
   546→                        var: var.clone(),
   547→                        line: line_of(node),
   548→                        rhs_kind,
   549→                        rhs_data: rhs_data.clone(),
   550→                    });
   551→                    if let Some(sk) = effects::classify_source(&rhs_text) {
   552→                        facts.sources.push(SourceSite {
   553→                            symbol: ctx.func,
   554→                            var,
   555→                            line: line_of(node),
   556→                            kind: sk,
   557→                        });
   558→                    }
   559→                }
   560→            }
   561→            walk_children(facts, ctx, node);
   562→        }
   563→        "string" => {
   564→            let t = text(node, ctx.bytes).to_string();
   565→            let is_fstring = t.starts_with("f\"")
   566→                || t.starts_with("f'")
   567→                || node
   568→                    .children(&mut node.walk())
   569→                    .any(|c| c.kind() == "interpolation");
   570→            if !is_fstring {
   571→                facts.literals.push(LiteralFact {
   572→                    symbol: ctx.func,
   573→                    line: line_of(node),
   574→                    value: t.chars().take(240).collect(),
   575→                    is_string: true,
   576→                });
   577→            } else {
   578→                facts.literals.push(LiteralFact {
   579→                    symbol: ctx.func,
   580→                    line: line_of(node),
   581→                    value: t.chars().take(240).collect(),
   582→                    is_string: false,
   583→                });
   584→            }
   585→            walk_children(facts, ctx, node);
   586→        }
   587→        "import_statement" => {
   588→            let mut c = node.walk();
   589→            for child in node.children(&mut c) {
   590→                match child.kind() {
   591→                    "dotted_name" | "identifier" => {
   592→                        facts.imports.push(ImportFact {
   593→                            module_symbol: ctx.module,
   594→                            module: text(child, ctx.bytes).to_string(),
   595→                            names: vec![],
   596→                            alias: None,
   597→                            line: line_of(node),
   598→                        });
   599→                    }
   600→                    "aliased_import" => {
   601→                        let name = child
   602→                            .child_by_field_name("name")
   603→                            .map(|n| text(n, ctx.bytes).to_string())
   604→                            .unwrap_or_default();
   605→                        let alias = child
   606→                            .child_by_field_name("alias")
   607→                            .map(|n| text(n, ctx.bytes).to_string());
   608→                        facts.imports.push(ImportFact {
   609→                            module_symbol: ctx.module,
   610→                            module: name,
   611→                            names: vec![],
   612→                            alias,
   613→                            line: line_of(node),
   614→                        });
   615→                    }
   616→                    _ => {}
   617→                }
   618→            }
   619→        }
   620→        "import_from_statement" => {
   621→            let module = node
   622→                .child_by_field_name("module_name")
   623→                .map(|n| text(n, ctx.bytes).to_string())
   624→                .unwrap_or_default();
   625→            let mut names = vec![];
   626→            let mut c = node.walk();
   627→            for child in node.children(&mut c) {
   628→                match child.kind() {
   629→                    "dotted_name" | "identifier" => names.push(text(child, ctx.bytes).to_string()),
   630→                    "aliased_import" => {
   631→                        if let Some(n) = child.child_by_field_name("name") {
   632→                            names.push(text(n, ctx.bytes).to_string());
   633→                        }
   634→                    }
   635→                    "wildcard_import" => names.push("*".into()),
   636→                    _ => {}
   637→                }
   638→            }
   639→            // "from . import x" — the relative import node is not module_name
   640→            let module = if module.is_empty() {
   641→                ".".into()
   642→            } else {
   643→                module
   644→            };
   645→            facts.imports.push(ImportFact {
   646→                module_symbol: ctx.module,
   647→                module,
   648→                names,
   649→                alias: None,
   650→                line: line_of(node),
   651→            });
   652→        }
   653→        "await" => {
   654→            facts.async_facts.push(AsyncSite {
   655→                symbol: ctx.func,
   656→                kind: AsyncKind::Await,
   657→                line: line_of(node),
   658→            });
   659→            walk_children(facts, ctx, node);
   660→        }
   661→        "pass_statement" | "break_statement" | "continue_statement" | "comment"
   662→        | "line_continuation" => {
   663→        }
   664→        _ => {
   665→            walk_children(facts, ctx, node);
   666→        }
   667→    }
   668→}

   670→fn classify_rhs(right: Node, bytes: &[u8], rhs_text: &str) -> (RhsKind, Vec<String>) {
   671→    match right.kind() {
   672→        "call" => {
   673→            let fn_text = right
   674→                .child_by_field_name("function")
   675→                .map(|n| text(n, bytes).to_string())
   676→                .unwrap_or_default();
   677→            if effects::classify_source(rhs_text).is_some() {
   678→                (RhsKind::Input, vec![fn_text])
   679→            } else {
   680→                (RhsKind::Call, vec![fn_text])
   681→            }
   682→        }
   683→        "string" => {
   684→            let t = rhs_text;
   685→            if t.starts_with("f\"") || t.starts_with("f'") {
   686→                let mut vars = vec![];
   687→                collect_interpolations(right, bytes, &mut vars);
   688→                (RhsKind::FString, dedup(vars))
   689→            } else {
   690→                (RhsKind::Literal, vec![])
   691→            }
   692→        }
   693→        "identifier" | "attribute" | "subscript" => {
   694→            let mut vars = vec![];
   695→            collect_identifiers(right, bytes, &mut vars);
   696→            (RhsKind::Vars, dedup(vars))
   697→        }
   698→        "binary_operator" | "boolean_operator" => {
   699→            let mut vars = vec![];
   700→            collect_identifiers(right, bytes, &mut vars);
   701→            (RhsKind::Vars, dedup(vars))
   702→        }
   703→        "integer" | "float" | "true" | "false" | "none" | "list" | "dictionary" | "set"
   704→        | "tuple" => (RhsKind::Literal, vec![]),
   705→        _ => {
   706→            let mut vars = vec![];
   707→            collect_identifiers(right, bytes, &mut vars);
   708→            if vars.is_empty() {
   709→                (RhsKind::Other, vec![])
   710→            } else {
   711→                (RhsKind::Vars, dedup(vars))
   712→            }
   713→        }
   714→    }
   715→}
   716→
   717→fn collect_interpolations(node: Node, bytes: &[u8], out: &mut Vec<String>) {
   718→    let mut c = node.walk();
   719→    for child in node.children(&mut c) {
   720→        if child.kind() == "interpolation" {
   721→            collect_identifiers(child, bytes, out);
   722→        } else if child.kind() == "string_content" {
   723→            // plain content — nothing
   724→        } else {
   725→            collect_interpolations(child, bytes, out);
   726→        }
   727→    }
   728→}
   729→
   730→fn collect_identifiers(node: Node, bytes: &[u8], out: &mut Vec<String>) {
   731→    match node.kind() {
   732→        "identifier" => {
   733→            let t = text(node, bytes).to_string();
   734→            if !t.is_empty() && t != "self" && t != "cls" {
   735→                out.push(t);
   736→            }
   737→        }
   738→        "attribute" => {
   739→            if let Some(attr) = node.child_by_field_name("attribute") {
   740→                out.push(text(attr, bytes).to_string());
   741→            }
   742→            if let Some(obj) = node.child_by_field_name("object") {
   743→                collect_identifiers(obj, bytes, out);
   744→            }
   745→        }
   746→        "keyword_argument" => {
   747→            if let Some(v) = node.child_by_field_name("value") {
   748→                collect_identifiers(v, bytes, out);
   749→            }
   750→        }
   751→        "comment" => (),
   752→        "string" | "concatenated_string" => {
   753→            // f-string interpolations carry taint
   754→            let mut c = node.walk();
   755→            let children: Vec<Node> = node.children(&mut c).collect();
   756→            for child in children {
   757→                if child.kind() == "interpolation" {
   758→                    collect_identifiers(child, bytes, out);
   759→                }
   760→            }
   761→        }
   762→        _ => {
   763→            // Document order (node.id() is not stable across parses — FR-4).
   764→            let mut c = node.walk();
   765→            for child in node.children(&mut c) {
   766→                collect_identifiers(child, bytes, out);
   767→            }
   768→        }
   769→    }
   770→}
   771→
   772→fn dedup(mut v: Vec<String>) -> Vec<String> {
   773→    v.sort();
   774→    v.dedup();
   775→    v
   776→}
   777→
   778→fn walk_children(facts: &mut LangFacts, ctx: &mut Ctx, node: Node) {
   779→    // children() yields document order; sorting by node.id() (the v0.1
   780→    // behavior) is NOT stable across parses and broke FR-4 byte-stable
   781→    // evidence — same source must always yield same facts.
   782→    let mut c = node.walk();
   783→    for child in node.children(&mut c) {
   784→        walk(facts, ctx, child);
   785→    }
   786→}
