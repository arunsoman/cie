    46→export function SeverityChip({ sev }: { sev: string }) {
    47→  const p = severityPlain(sev);
    48→  const cls =
    49→    p.tone === "rose"
    50→      ? "bg-rose-100 text-rose-800 border-rose-200"
    51→      : p.tone === "amber"
    52→        ? "bg-amber-100 text-amber-800 border-amber-200"
    53→        : "bg-stone-100 text-stone-700 border-stone-200";
    54→  return <Badge variant="outline" className={`${cls} font-semibold`}>{p.label}</Badge>;
    55→}
    56→
    57→export function CertaintyChip({ mode }: { mode: string }) {
    58→  const c = certaintyPlain(mode);
    59→  const cls =
    60→    mode === "Fact"
    61→      ? "bg-emerald-50 text-emerald-800 border-emerald-200"
    62→      : mode === "Inference"
    63→        ? "bg-amber-50 text-amber-800 border-amber-200"
    64→        : "bg-stone-100 text-stone-700 border-stone-200";
    65→  return <Badge variant="outline" className={cls}>{c.label}</Badge>;
    66→}

    68→function CoverageRing({ pct }: { pct: number }) {
    69→  const r = 34, c = 2 * Math.PI * r;
    70→  const filled = (Math.min(100, Math.max(0, pct)) / 100) * c;
    71→  return (
    72→    <svg width="88" height="88" viewBox="0 0 88 88" role="img" aria-label={`Checked: ${pct}%`}>
    73→      <circle cx="44" cy="44" r={r} fill="none" stroke="#f5f5f4" strokeWidth="9" />
    74→      <circle
    75→        cx="44" cy="44" r={r} fill="none" stroke="#059669" strokeWidth="9"
    76→        strokeLinecap="round" strokeDasharray={`${filled} ${c - filled}`}
    77→        transform="rotate(-90 44 44)"
    78→      />
    79→      <text x="44" y="49" textAnchor="middle" fontSize="17" fontWeight="700" fill="#1c1917">
    80→        {Math.round(pct)}%
    81→      </text>
    82→    </svg>
    83→  );
    84→}

    86→function ruleKindPlain(k?: string): string {
    87→  if (!k) return "Rule";
    88→  const map: Record<string, string> = {
    89→    must_authorize: "Check permission first",
    90→    must_not_leak_secrets: "Keep secrets secret",
    91→    must_validate_input: "Check user input before using it",
    92→  };
    93→  return map[k] ?? k.replace(/_/g, " ");
    94→}
    95→
    96→function ruleStatusPlain(s?: string): { label: string; cls: string } {
    97→  const v = (s ?? "").toLowerCase();
    98→  if (v === "explicit") return { label: "Written by you", cls: "bg-emerald-50 text-emerald-800 border-emerald-200" };
    99→  if (v === "inferred_candidate" || v === "inferredcandidate") return { label: "CIE's guess", cls: "bg-amber-50 text-amber-800 border-amber-200" };
   100→  return { label: s ?? "", cls: "bg-stone-100 text-stone-700 border-stone-200" };
   101→}

   272→export function ConceptsView() {
   273→  const { concepts, report } = useCie();
   274→  if (!concepts || !report) return null;
   275→  const roots = concepts.nodes.filter((n) => n.parent === null);
   276→  const modules = concepts.nodes.filter((n) => n.kind === "Module");
   277→  const symName = (id: number) => report.symbols.find((s) => s.id === id)?.qual_name ?? "?";
   278→
   279→  const ruleList = (node: (typeof concepts.nodes)[number]) => (
   280→    <>
   281→      {node.behaviors.length > 0 && (
   282→        <div className="flex flex-wrap gap-1.5 mb-2">
   283→          {node.behaviors.map((b, i) => (
   284→            <Badge key={i} variant="outline" className="bg-emerald-50 text-emerald-800 border-emerald-200 font-normal">
   285→              Does: {b.kind}
   286→            </Badge>
   287→          ))}
   288→        </div>
   289→      )}
   290→      {node.rules && node.rules.length > 0 && (
   291→        <div className="flex flex-wrap gap-1.5">
   292→          {node.rules.map((r, i) => {
   293→            const st = ruleStatusPlain(String(r.authority ?? "").toLowerCase());
   294→            const rid = typeof r.contract === "number" ? `C-${String(r.contract).padStart(3, "0")}` : "";
   295→            return (
   296→              <Badge key={i} variant="outline" className={`${st.cls} font-normal`}>
   297→                Rule {rid}: {ruleKindPlain(r.kind)} · {st.label}
   298→              </Badge>
   299→            );
   300→          })}
   301→        </div>
   302→      )}
   303→    </>
   304→  );
   305→
   306→  return (
   307→    <div className="space-y-4">
   308→      <Card className="border-stone-200">
   309→        <CardHeader className="pb-3">
   310→          <CardTitle className="text-base">The big picture</CardTitle>
   311→          <CardDescription>What your whole project does, in one glance.</CardDescription>
   312→        </CardHeader>
   313→        <CardContent>
   314→          {roots.map((r) => (
   315→            <div key={r.id}>
   316→              {ruleList(r)}
   317→              <p className="text-xs text-stone-500 mt-2">
   318→                Covers {r.symbols.length} named things, e.g. {r.symbols.slice(0, 4).map(symName).join(", ")}
   319→                {r.symbols.length > 4 ? "…" : ""}
   320→              </p>
   321→            </div>
   322→          ))}
   323→        </CardContent>
   324→      </Card>
   325→
   326→      <div className="grid gap-4 md:grid-cols-2">
   327→        {modules.map((m) => (
   328→          <Card key={m.id} className="border-stone-200">
   329→            <CardHeader className="pb-2">
   330→              <CardTitle className="text-base font-mono">{m.label}</CardTitle>
   331→              <CardDescription>A code file · {m.symbols.length} named things inside</CardDescription>
   332→            </CardHeader>
   333→            <CardContent>{ruleList(m)}</CardContent>
   334→          </Card>
   335→        ))}
   336→      </div>
   337→    </div>
   338→  );
   339→}

   379→export function FindingsView() {
   380→  const { report } = useCie();
   381→  const sorted = useMemo(() => {
   382→    if (!report) return [];
   383→    const order = { Critical: 0, High: 1, Medium: 2, Low: 3 } as Record<string, number>;
   384→    return [...report.findings].sort((a, b) => (order[a.severity] ?? 9) - (order[b.severity] ?? 9));
   385→  }, [report]);
   386→  if (!report) return null;
   387→
   388→  const groups: { key: string; title: string; desc: string }[] = [
   389→    { key: "Critical", title: "Fix soon", desc: "These could let the wrong people in or lose data." },
   390→    { key: "High", title: "Please look", desc: "Not an emergency, but worth your attention." },
   391→    { key: "Medium", title: "Worth checking", desc: "Could cause trouble later." },
   392→    { key: "Low", title: "Good to know", desc: "No rush — just so you know." },
   393→  ];
   394→
   395→  if (!report.findings.length) {
   396→    return (
   397→      <div>
   398→        <Card className="border-emerald-200 bg-emerald-50">
   399→          <CardContent className="p-8 text-center">
   400→            <BadgeCheck className="h-10 w-10 text-emerald-700 mx-auto mb-3" />
   401→            <p className="font-semibold text-emerald-900 text-lg">Nothing to worry about</p>
   402→            <p className="text-sm text-emerald-800 mt-1">CIE looked through everything and found no problems to report. Well done!</p>
   403→          </CardContent>
   404→        </Card>
   405→      </div>
   406→    );
   407→  }
   408→
   409→  return (
   410→    <div className="space-y-6">
   411→      {groups.map((g) => {
   412→        const items = sorted.filter((f) => f.severity === g.key);
   413→        if (!items.length) return null;
   414→        return (
   415→          <section key={g.key} aria-label={g.title} className="space-y-3">
   416→            <div>
   417→              <h3 className="font-semibold text-stone-900">{g.title} <span className="text-stone-400 font-normal">({items.length})</span></h3>
   418→              <p className="text-sm text-stone-500">{g.desc}</p>
   419→            </div>
   420→            {items.map((f) => (
   421→              <FindingCard key={f.id} f={f} file={report.files[f.symbol ? report.symbols.find((s) => s.id === f.symbol)?.loc.file ?? 0 : 0] ?? ""} />
   422→            ))}
   423→          </section>
   424→        );
   425→      })}
   426→    </div>
   427→  );
   428→}
