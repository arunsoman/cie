    68→    68→function CoverageRing({ pct }: { pct: number }) {
    69→    69→  const r = 34, c = 2 * Math.PI * r;
    70→    70→  const filled = (Math.min(100, Math.max(0, pct)) / 100) * c;
    71→    71→  return (
    72→    72→    <svg width="88" height="88" viewBox="0 0 88 88" role="img" aria-label={`Checked: ${pct}%`}>
    73→    73→      <circle cx="44" cy="44" r={r} fill="none" stroke="#f5f5f4" strokeWidth="9" />
    74→    74→      <circle
    75→    75→        cx="44" cy="44" r={r} fill="none" stroke="#059669" strokeWidth="9"
    76→    76→        strokeLinecap="round" strokeDasharray={`${filled} ${c - filled}`}
    77→    77→        transform="rotate(-90 44 44)"
    78→    78→      />
    79→    79→      <text x="44" y="49" textAnchor="middle" fontSize="17" fontWeight="700" fill="#1c1917">
    80→    80→        {Math.round(pct)}%
    81→    81→      </text>
    82→    82→    </svg>
    83→    83→  );
    84→    84→}

    86→    86→function ruleKindPlain(k?: string): string {
    87→    87→  if (!k) return "Rule";

    92→    92→  };
    93→    93→  return map[k] ?? k.replace(/_/g, " ");

    96→    96→function ruleStatusPlain(s?: string): { label: string; cls: string } {
    97→    97→  const v = (s ?? "").toLowerCase();
    98→    98→  if (v === "explicit") return { label: "Written by you", cls: "bg-emerald-50 text-emerald-800 border-emerald-200" };
    99→    99→  if (v === "inferred_candidate" || v === "inferredcandidate") return { label: "CIE's guess", cls: "bg-amber-50 text-amber-800 border-amber-200" };
